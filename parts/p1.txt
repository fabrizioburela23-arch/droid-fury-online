let flow=null, flowDirty=true;
function computeFlow(){ const W=MAP_W,H=MAP_H; const dist=new Int16Array(W*H).fill(-1); const q=[];
 for(const pl of P){ if(pl.out||pl.downed) continue; const tx=clamp((pl.x/TILE)|0,0,W-1),ty=clamp((pl.y/TILE)|0,0,H-1); const i=ty*W+tx; if(dist[i]<0){ dist[i]=0; q.push(i); } }
 let h=0; while(h<q.length){ const i=q[h++],x=i%W,y=(i/W)|0,d=dist[i]+1; if(x+1<W&&grid[y][x+1]===0&&dist[i+1]<0){ dist[i+1]=d; q.push(i+1); } if(x>0&&grid[y][x-1]===0&&dist[i-1]<0){ dist[i-1]=d; q.push(i-1); } if(y+1<H&&grid[y+1][x]===0&&dist[i+W]<0){ dist[i+W]=d; q.push(i+W); } if(y>0&&grid[y-1][x]===0&&dist[i-W]<0){ dist[i-W]=d; q.push(i-W); } }
 flow=dist; flowDirty=false; }
function steer(z,tgt){ const dx=tgt.x-z.x,dy=tgt.y-z.y,d=Math.hypot(dx,dy)||1;
 if(d<320&&hasLOS(z.x,z.y,tgt.x,tgt.y)) return {x:dx/d,y:dy/d,d};
 if(flow){ const tx=clamp((z.x/TILE)|0,0,MAP_W-1),ty=clamp((z.y/TILE)|0,0,MAP_H-1); let best=null; let bd=flow[ty*MAP_W+tx]; if(bd<0) bd=30000;
  for(let oy=-1;oy<=1;oy++) for(let ox=-1;ox<=1;ox++){ if(!ox&&!oy) continue; const nx=tx+ox,ny=ty+oy; if(nx<0||ny<0||nx>=MAP_W||ny>=MAP_H||grid[ny][nx]) continue; if(ox&&oy&&(grid[ty][nx]||grid[ny][tx])) continue; const v=flow[ny*MAP_W+nx]; if(v>=0&&v<bd){ bd=v; best={nx,ny}; } }
  if(best){ const cx=(best.nx+0.5)*TILE,cy=(best.ny+0.5)*TILE,ex=cx-z.x,ey=cy-z.y,ed=Math.hypot(ex,ey)||1; return {x:ex/ed,y:ey/ed,d}; } }
 return {x:dx/d,y:dy/d,d}; }
