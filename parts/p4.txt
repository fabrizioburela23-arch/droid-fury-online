function update(dt){ G.time+=dt; G.stats.time+=dt;
 G.flowT-=dt; if(flowDirty||G.flowT<=0){ computeFlow(); G.flowT=0.2; }

let last=0;
function loop(ts){ requestAnimationFrame(loop); let dt=Math.min(0.05,(ts-last)/1000)||0; last=ts; pollGamepadMenus(); updateProbe(); TOUCH.enabled=touchActive(); if(!TOUCH.enabled){ TOUCH.left=TOUCH.right=null; TOUCH.held={}; }

 else if(G.state==='upgrade'||G.state==='dead'){ updateFx(dt); if(G.state==='dead') G.time+=dt; }
 render(dt); clearEdges(); }
